# 바닐라 호환용 스크립트 트리거
# 이 모드는 common/scripted_triggers를 replace_path로 통째로 교체하기 때문에,
# 교체되지 않은 바닐라 파일(achievements, decision categories, dynamic modifiers, music 등)이
# 참조하는 바닐라 트리거들이 사라져 "Unknown trigger-type" 에러가 발생함.
# 해당 바닐라 콘텐츠는 이 모드에서 사용되지 않으므로 대부분 always = no 스텁으로 정의.
# 개별 국가 컨텐츠를 추가할 때 필요하면 바닐라 정의로 교체할 것.

# --- 다르다넬스 해협 통행 규칙 (map/adjacency_rules.txt에서 실제 사용됨 - 간략화 버전) ---
TUR_is_friend_for_dardanelles = {
	hidden_trigger = {
		OR = {
			AND = { # 튀르키예가 해협 통제 시: 튀르키예와 전쟁 중이 아니면 통행 가능
				TUR = {
					controls_province = 6864
					controls_province = 849
				}
				NOT = { has_war_with = TUR }
			}
			797 = { # 해협 통제국의 동맹/군사통행권 보유국도 통행 가능
				controller = {
					OR = {
						tag = ROOT
						is_in_faction_with = ROOT
						ROOT = { has_military_access_to = PREV }
					}
				}
			}
		}
	}
}
